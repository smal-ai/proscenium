def safe(path):
    p = Path(path)
    if not p.is_absolute():
        p = root / p
    assert p.is_relative_to(root) and not any(x.is_symlink() for x in [p, *p.parents]), str(p)
    return p


def sha(path):
    return hashlib.sha256(safe(path).read_bytes()).hexdigest()


def plain(value):
    if isinstance(value, (str, int, float, bool)) or value is None:
        return value
    if hasattr(value, "to_dict"):
        return {k: plain(v) for k, v in value.to_dict().items()}
    return [plain(v) for v in value]


def ids(data):
    return {k: plain(v) for k, v in data.items()}


def cubes():
    return [o for o in bpy.context.scene.objects if o.type == "MESH" and (
        o.data.name.split(".")[0] == "Cube" or
        o.get("object_id") == "object-cube-source" or
        o.get("proscenium_object_id") == "object-cube-source" or
        o.data.get("object_id") == "object-cube-source" or
        o.data.get("proscenium_object_id") == "object-cube-source"
    )]


def measured(o):
    points = [o.matrix_world @ Vector(c) for c in o.bound_box]
    lo = [min(p[i] for p in points) for i in range(3)]
    hi = [max(p[i] for p in points) for i in range(3)]
    return {
        "name": o.name, "ids": ids(o), "mesh": o.data.name, "mesh_ids": ids(o.data),
        "library": o.data.library.filepath if o.data.library else None,
        "bounds_m": {"min": lo, "max": hi}, "dimensions_m": [hi[i] - lo[i] for i in range(3)],
        "location": list(o.location), "materials": [s.material.name if s.material else None for s in o.material_slots],
        "vertices": len(o.data.vertices), "polygons": len(o.data.polygons),
    }


def dependencies(scope):
    paths = list(bpy.utils.blend_paths(absolute=True, packed=False, local=False))
    for path in paths:
        p = safe(path)
        assert p.is_relative_to(scope) and p.is_file(), f"Required path outside inspected source root or missing: {p}"
    libraries = []
    for lib in bpy.data.libraries:
        p = safe(bpy.path.abspath(lib.filepath))
        assert p.is_relative_to(scope) and p.is_file()
        libraries.append({"name": lib.name, "filepath": lib.filepath, "resolved_path": str(p), "sha256": sha(p)})
    textures = []
    for image in bpy.data.images:
        if image.source != "FILE" or not image.filepath or image.packed_file:
            continue
        p = safe(bpy.path.abspath(image.filepath, library=image.library))
        assert p.is_relative_to(scope) and p.is_file()
        textures.append({"name": image.name, "filepath": image.filepath, "resolved_path": str(p), "dimensions": list(image.size), "sha256": sha(p)})
    return {"blend_paths": paths, "libraries": libraries, "textures": textures}


def animation(o):
    data = o.animation_data
    if not data:
        return None
    action = data.action
    curves = []
    if action:
        for curve in action.fcurves:
            curves.append({"path": curve.data_path, "index": curve.array_index, "keys": [
                {"co": list(k.co), "left": list(k.handle_left), "right": list(k.handle_right), "interpolation": k.interpolation}
                for k in curve.keyframe_points
            ]})
    return {"action": action.name if action else None, "curves": curves,
            "drivers": [(f.data_path, f.array_index, f.driver.expression) for f in data.drivers],
            "nla": [(t.name, [(s.name, s.action.name if s.action else None, s.frame_start, s.frame_end) for s in t.strips]) for t in data.nla_tracks]}


def scene_state():
    scene = bpy.context.scene
    return {
        "objects": {o.name: {"ids": ids(o), "location": list(o.location), "rotation": list(o.rotation_euler),
                             "scale": list(o.scale), "animation": animation(o)} for o in scene.objects},
        "camera": scene.camera.name if scene.camera else None,
        "timeline": [scene.frame_start, scene.frame_end, scene.frame_current, scene.render.fps, scene.render.fps_base],
    }
